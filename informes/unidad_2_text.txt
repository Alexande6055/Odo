¿Qué es la calidad del producto de software?

La calidad del producto de software es el grado en que un sistema cumple con:
• Requisitos funcionales
• Requisitos no funcionales
• Expectativas del usuario
• Condiciones de uso reales

Un software es de calidad cuando funciona bien, no falla, es fácil de usar y resuelve el problema correctamente.

from odoo.tests.common import TransactionCase
class TestMathOperationTime(TransactionCase):
def setUp(self):
super().setUp()
self.MathOperation = self.env['math.operations']
def test_division_response_time(self):
inicio = time. perf_counter()
rec = self.MathOperation.create({
'operand_1': 100,
'operand_2': 5,
'operation_type': 'division'
})
fin = time. perf_counter()
tiempo = fin - inicio
self.assertEqual(rec.result, 20)
self.assertLess(tiempo, 1) # debe responder en menos de 1 segundo
print(f"Tiempo de respuesta: {tiempo:.4f} segundos")
